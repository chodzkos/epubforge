167        """Otwiera archiwum do odczytu i waliduje, że to EPUB.

169        Raises:

171            OpfNotFoundError: brak ``META-INF/container.xml``.

173        if self._zip is not None:
174            return
175        if not self.path.is_file():
176            raise InvalidEpubError(f"Plik EPUB nie istnieje: {self.path}")

178            zf = zipfile.ZipFile(self.path)

182            zf.close()
183            raise OpfNotFoundError(f"Brak {_CONTAINER_PATH} — to nie jest poprawny EPUB")
184        self._zip = zf
185        logger.debug("Otwarto EPUB: %s", self.path)

252    def read_file(self, internal_path: str) -> bytes:
253        """Zwraca zawartość pliku wewnątrz EPUB-a.
254
255        Uwzględnia niezapisane zmiany z :meth:`write_file`.
256
257        Args:
258            internal_path: ścieżka względna wewnątrz archiwum (z ``/``).
259
260        Raises:
261            EpubNotOpenError: gdy EPUB nie jest otwarty.
262            KeyError: gdy plik nie istnieje w archiwum.
263        """
264        zf = self._ensure_open()
265        if internal_path in self._deleted:
266            raise KeyError(internal_path)
267        if internal_path in self._modified:
268            return self._modified[internal_path]
269        return zf.read(internal_path)

271    def write_file(self, internal_path: str, data: bytes) -> None:
272        """Zapisuje zmianę pliku do bufora w pamięci (utrwalane w :meth:`save`).
273
274        Args:
275            internal_path: ścieżka względna wewnątrz archiwum (z ``/``).

277
278        Raises:
279            EpubNotOpenError: gdy EPUB nie jest otwarty.
280        """
281        self._ensure_open()
282        self._deleted.discard(internal_path)
283        self._modified[internal_path] = data
284        # Modyfikacja OPF unieważnia zcache'owany manifest/spine.
285        if internal_path == self._opf_path:
286            self._manifest = None
287            self._spine = None

290        """Oznacza plik wewnętrzny do usunięcia przy następnym :meth:`save`.

324    def save(self, output_path: Path | None = None) -> Path:
325        """Zapisuje EPUB z poprawną strukturą ZIP.
326
327        Args:
328            output_path: gdy ``None`` — nadpisuje oryginał (po wykonaniu
329                :meth:`backup`); w przeciwnym razie zapisuje pod wskazaną
330                ścieżką, a bieżąca sesja pozostaje otwarta na oryginale.
331
332        Returns:
333            Ścieżka faktycznie zapisanego pliku.
334
335        Raises:
336            EpubNotOpenError: gdy EPUB nie jest otwarty.
337        """
338        self._ensure_open()
339        modified = dict(self._modified)
340        deleted = set(self._deleted)
341        if output_path is not None:
342            target = Path(output_path)
343            _write_epub(self.path, target, modified, deleted)
344            logger.debug("Zapisano EPUB jako: %s", target)
345            return target
346        # Nadpisanie oryginału: backup, zamknięcie uchwytu (Windows), zapis, reopen.
347        self.backup()
348        assert self._zip is not None
349        self._zip.close()
350        self._zip = None
351        _write_epub(self.path, self.path, modified, deleted)
352        self._modified.clear()
353        self._deleted.clear()
354        self._reset_cache()
355        self.open()
356        logger.debug("Nadpisano EPUB: %s", self.path)
357        return self.path

372    def _ensure_open(self) -> zipfile.ZipFile:
373        """Zwraca otwarty uchwyt ZIP albo rzuca :class:`EpubNotOpenError`."""
374        if self._zip is None:
375            raise EpubNotOpenError("EPUB nie jest otwarty — wywołaj open() lub użyj 'with'.")
376        return self._zip

378    def _reset_cache(self) -> None:
379        """Czyści zcache'owaną ścieżkę OPF, manifest i spine."""
380        self._opf_path = None
381        self._manifest = None

384    def _parse_opf(self) -> None:
385        """Parsuje plik OPF i wypełnia cache manifestu oraz spine."""
386        try:
387            root = parse_untrusted(self.read_file(self.opf_path))
388        except etree.XMLSyntaxError as exc:
389            raise OpfNotFoundError(f"Niepoprawny plik OPF ({self.opf_path}): {exc}") from exc
390
391        manifest: list[ManifestItem] = []
392        for item in root.iterfind(f"{{{_OPF_NS}}}manifest/{{{_OPF_NS}}}item"):
393            item_id = item.get("id")
394            href = item.get("href")
395            media_type = item.get("media-type")
396            if item_id is None or href is None or media_type is None:
397                continue
398            manifest.append(
399                ManifestItem(
400                    id=item_id,
401                    href=href,
402                    media_type=media_type,
403                    properties=item.get("properties"),
404                )
405            )
406
407        spine: list[str] = []
408        for itemref in root.iterfind(f"{{{_OPF_NS}}}spine/{{{_OPF_NS}}}itemref"):
409            idref = itemref.get("idref")
410            if idref is not None:
411                spine.append(idref)
412
413        self._manifest = manifest
414        self._spine = spine

416    def opf_dir(self) -> str:
417        """Zwraca katalog pliku OPF (przydatny do rozwiązywania ``href``)."""
418        return posixpath.dirname(self.opf_path)

210    def to_opf(self, existing_opf: bytes) -> bytes:
211        """Wstrzykuje metadane do istniejącego OPF, zachowując resztę dokumentu.
212
213        Sekcje ``manifest`` i ``spine`` oraz nieznane elementy nie są ruszane.
214        Pola wielowartościowe (autorzy, tematy) są zastępowane w całości;
215        pola pojedyncze są aktualizowane w miejscu (z zachowaniem atrybutów,
216        np. ``id`` przy ``dc:identifier``).
217
218        Args:
219            existing_opf: surowa zawartość obecnego pliku OPF.
220
221        Returns:
222            Nowa zawartość OPF jako bajty UTF-8, z deklaracją XML.
223        """
224        root = parse_untrusted(existing_opf)
225        metadata_el = root.find(f"{{{OPF_NS}}}metadata")
226        if metadata_el is None:
227            metadata_el = etree.SubElement(root, f"{{{OPF_NS}}}metadata")
228            root.insert(0, metadata_el)
229
230        # Pola pojedyncze — aktualizacja w miejscu (lub utworzenie gdy brak).
231        for attr_name, tag in _SINGLE_FIELDS:
232            value = getattr(self, attr_name)
233            self._set_single(metadata_el, tag, value)
234
235        # Pola wielowartościowe — zastąp wszystkie wystąpienia.
236        self._set_multi(metadata_el, "creator", self.creators)
237        self._set_multi(metadata_el, "subject", self.subjects)
238
239        # Seria — Calibre (zawsze) + EPUB 3 (gdy package version 3.x).
240        epub3 = str(root.get("version", "")).startswith("3")
241        self._set_series(metadata_el, epub3=epub3)
242
243        return etree.tostring(root, xml_declaration=True, encoding="utf-8")

245    @staticmethod
246    def _set_single(metadata_el: etree._Element, tag: str, value: str) -> None:
247        """Ustawia tekst pierwszego ``dc:<tag>`` (tworzy element gdy brak)."""
248        if not value:
249            return
250        el = metadata_el.find(_dc(tag))
251        if el is None:
252            el = etree.SubElement(metadata_el, _dc(tag))
253        el.text = value
254
255    @staticmethod
256    def _set_multi(metadata_el: etree._Element, tag: str, values: list[str]) -> None:
257        """Usuwa istniejące ``dc:<tag>`` i wstawia po jednym na każdą wartość."""
258        for el in metadata_el.findall(_dc(tag)):
259            metadata_el.remove(el)
260        for value in values:
261            etree.SubElement(metadata_el, _dc(tag)).text = value
262
263    def _set_series(self, metadata_el: etree._Element, *, epub3: bool) -> None:
264        """Zapisuje serię (Calibre + opcjonalnie EPUB 3); pusta seria = usunięcie.
265
266        Najpierw usuwa wszystkie istniejące meta serii (oba formaty), żeby zapis
267        był idempotentny i nie zostawiał osieroconych wpisów.
268        """
269        self._remove_series(metadata_el)
270        if not self.series:
271            return
272
273        index_text = "" if self.series_index is None else _format_index(self.series_index)
274
275        # Wariant Calibre — zawsze (kompatybilność z wieloma czytnikami).
276        etree.SubElement(metadata_el, _opf("meta"), name="calibre:series", content=self.series)
277        if index_text:
278            etree.SubElement(
279                metadata_el, _opf("meta"), name="calibre:series_index", content=index_text
280            )
281
282        # Wariant EPUB 3 — tylko dla pakietów 3.x.
283        if epub3:
284            collection = etree.SubElement(metadata_el, _opf("meta"))
285            collection.set("property", "belongs-to-collection")
286            collection.set("id", _COLLECTION_ID)
287            collection.text = self.series
288
289            ctype = etree.SubElement(metadata_el, _opf("meta"))
290            ctype.set("refines", f"#{_COLLECTION_ID}")
291            ctype.set("property", "collection-type")
292            ctype.text = "series"
293
294            if index_text:
295                position = etree.SubElement(metadata_el, _opf("meta"))
296                position.set("refines", f"#{_COLLECTION_ID}")
297                position.set("property", "group-position")
298                position.text = index_text
299
300    @staticmethod
301    def _remove_series(metadata_el: etree._Element) -> None:
302        """Usuwa wszystkie meta serii (Calibre i EPUB 3) z sekcji metadanych."""
303        for meta in list(metadata_el.findall(_opf("meta"))):
304            name = meta.get("name")
305            prop = meta.get("property")
306            is_calibre = name in {"calibre:series", "calibre:series_index"}
307            is_epub3 = prop in {"belongs-to-collection", "collection-type", "group-position"}
308            if is_calibre or is_epub3:
309                metadata_el.remove(meta)
